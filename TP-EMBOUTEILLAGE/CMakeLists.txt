cmake_minimum_required(VERSION 3.20)
project(tp2_embouteillage CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
set(AVERTISSEMENTS -Wall -Wextra -Wpedantic -Wconversion -Wno-sign-conversion)
find_package(Threads REQUIRED)

add_executable(ligne src/ligne.cpp)
target_compile_options(ligne PRIVATE ${AVERTISSEMENTS})
target_link_libraries(ligne PRIVATE Threads::Threads)

add_executable(course src/course.cpp)
target_compile_options(course PRIVATE ${AVERTISSEMENTS})
target_link_libraries(course PRIVATE Threads::Threads)

enable_testing()
add_executable(test_file tests/test_file.cpp)
target_compile_options(test_file PRIVATE ${AVERTISSEMENTS} -fsanitize=thread)
target_link_options(test_file PRIVATE -fsanitize=thread)
target_link_libraries(test_file PRIVATE Threads::Threads)
add_test(NAME file_bornee COMMAND test_file)
